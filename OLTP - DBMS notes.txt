UNION | INTERSET | EXCEPT
RULE: Table structure should be same.

UNION: 
 It combine rows from multiple table
 It gives all record from both the table
 syntax
	SELECT * FROM table1 UNION SELECT * FROM  table2

INTERSET: 
 It gives only common records from both table
 syntax
	SELECT * FROM table1 INTERSET SELECT * FROM  table2

EXCEPT: 
 It gives records that are not common in both table
 syntax
	SELECT * FROM table1 EXCEPT SELECT * FROM  table2

CREATE OR REPLACE PROCEDURE multiplaction(num1 int )
language plpgsql
as $$
declare
	n1 int :=1;
	val1 int :=0;
begin
While (n1 <= 10) loop
	val1 = num1 * n1;
	raise notice '% * % = %', num1,n1,val1;
	n1 := n1+1;
end loop;
end;$$

query life cycles or query execution flow
------------------------------------------------------
1. parsing the query - syntax
2. parsing the object - semantics (it will check whatever are the object we uses in query is present in db or not (like emp , dept )) 
3. query optimizer - best way to execute the query
    create plans and choose best way to execute the query
4. query + plans = storage engines( their are two types of engines - relation and storage engines - relation engine is responsible for making relation 
between tables and storage engines is responsible for managing storage) 
5. result is given to the client using network

CREATE OR REPLACE PROCEDURE sp_calcu()
language plpgsql
as $$
declare 
	t int;
begin
	t := n1 /n2;
	raise notice 'total = %', t;
exception
	when others then 
		raise notice 'exception occured at %', SQLSTATE;
end; $$

-- FOR EXCPTION LOGING


CREATE TABLE ERRORLOGS (
ERRORID INT GENERATED ALWAYS AS IDENTITY (START WITH 1 INCREMENT BY 1),
ERRORMSG VARCHAR (2000),
ERRORSTATE VARCHAR (2000),
ERRORGENERATEDBY VARCHAR (2000),
ERRORDATETIME TIMESTAMP
);

CREATE OR REPLACE PROCEDURE gen_log(n1 int, n2 int)
language plpgsql
as $$
declare 
	t int;
begin
	begin
	t := n1 /n2;
	raise notice 'total = %', t;
	exception
	when others then 
		raise notice 'exception occured at %', SQLSTATE;
				INSERT INTO ERRORLOGS(ERRORMSG,ERRORSTATE,ERRORGENERATEDBY,ERRORDATETIME) VALUES (SQLERRM,SQLSTATE,CURRENT_USER,NOW());
	END;
	 
	t := n1+n2;
	exception
	when others then 
		raise notice 'exception occured at %', SQLSTATE;
		INSERT INTO ERRORLOGS(ERRORMSG,ERRORSTATE,ERRORGENERATEDBY,ERRORDATETIME) VALUES (SQLERRM,SQLSTATE,CURRENT_USER,NOW());
end; $$

call gen_log (10,0);

# Index
- it help to optimize query and increase performance

SYNTAX 
CREATE INDEX index_name ON table_name (column_name)
----------------------------------------------------
TYPES OF INDEXES
1} clustred index / btree index (default index, main index)
-> this is the primary index of the table
	CREATE INDEX index_name ON table_name (column_name)

2} Multi - column index
-> The index will be created on multi columns 
	CREATE INDEX index_name ON table_name (column1,column2,column3)


3} Unique index
-> it should be make on unique column 
	CREATE UNIQUE INDEX index_name ON table_name (bookid)

4} Partial Index
-> partial index will create only on certain column values of the table 
	CREATE UNIQUE INDEX index_name ON table_name (bookid) WHERE condition

5} Hash Index
-> It use hash model to store values, it dose not use tree structure followed by B-TREE
	CREATE INDEX index_name ON table_name USING HASH (COLUMN_NAME);

# Cursor

it is the pointer to the record set or query using cursor, we can iterate the records and read the columns etc. and do operations on that 
1} Declare cursor
2} open cursor
3} Fetch
4} iteration through record using loop
5} close cursor 

Disadvantage
1 it consume lots of memory


SYNTAX 
do
$$
declare 
	cur_emp CURSOR FOR SELECT emp_id, emp_name FROM employees;
	eid int;
	ename varchar(100);

Begin
	OPEN cur_emp;
	fetch next from cur_emp into eid,ename;
	loop
	raise notice 'ID = %  Name = %',eid,ename;
	fetch next from cur_emp into eid,ename;
	exit when not found;
	end loop;
end;$$
